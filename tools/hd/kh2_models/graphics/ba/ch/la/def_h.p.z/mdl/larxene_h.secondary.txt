# Secondary motion of KH2's Larxene (B_EX400) drawn as Days' larxene_h and larxene: marluxia_h's coat, drawstrings and
# hood on her bones (the same numbers), and her short hair. Bones are KH2 bone numbers (kh2_<n>). docs/SECONDARY_MOTION.md
# explains every line. Her cuffs are fitted in bones.txt (`cuff`); her antennae's strands follow Days' hair nodes there.
secondary_motion 1
units cm
include kh2_secondary_presets.txt

# 60 Hz; 3 substeps of 2 passes, as roxas_bf_h
simulation rate 60 substeps 3 iterations 2

teleport 150 100            # a jump of 1.5 m or 100 degrees in one tick rebases

# --- Drawstrings: chains from the collar (97), hung at load: they stick straight forward at rest ---
group string_l chain preset kh2_string
  # the arms too, and damped: the tips no longer whip when the hands cross the chest

  attach 97
  chain 101 102 103 104
  hang 0 -1 0.35
  settle on steps 360

# --- Animation controls ---
# B_EX400's own motions carry its coat and drawstrings as KH2 animated them: there they guide the cloth, and the thighs
# carry none of it. The game's own animations keep the cloth on the hips and thighs.
control own_motions
  animation *
  slot 0-255
  groups coat string_l string_r
  mode guided
  follow_frequency 6 hem 3
  follow_damping 0.8
  follow_bone_scale 0
  blend 0.15 0.15
end
