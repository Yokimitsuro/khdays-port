# Secondary motion of KH2's Xemnas (B_EX170) drawn as Days' xemnas_h and xemnas: axel_h's coat on the same bones, his
# drawstrings, hood and hair. Bones are KH2 bone numbers (kh2_<n>). docs/SECONDARY_MOTION.md explains every line. His
# cuffs are fitted in bones.txt (`cuff`), not simulated.
secondary_motion 1
units cm
include kh2_secondary_presets.txt

# 60 Hz; 3 substeps of 2 passes, as roxas_bf_h
simulation rate 60 substeps 3 iterations 2
gravity 0 -9.81 0           # m/s^2, the world's y up
environment on thickness 2 ground  # the room's floors alone
teleport 150 100            # a jump of 1.5 m or 100 degrees in one tick rebases
first_pose_iterations 30

# --- Body colliders (KH2 bones): `fit` sizes itself to the vertices skinned to them. His legs and hands are demyx_h's
# vertex for vertex, so the fixed sizes are his (axel_h's); the pelvis, chest and thighs by his own bones' lengths. ---
collider pelvis   capsule 2 length 29.7 fit 0.85 tags body
collider spine    capsule 96 98 fit 0.85 tags torso
# 20 of the 26.1 from 98 to the collar (99), as demyx_h's 18 of 23.2: at its full length its end reaches the hood
collider chest    capsule 98 length 20 fit 0.85 tags torso
collider thigh_l  capsule 5 6 offset 17.6 0 0 fit 0.9 tags legs
collider calf_l   capsule 6 9 fit 0.9 tags legs
collider foot_l   capsule 9 length 11.7 radius 3.5 2.7 tags legs
collider thigh_r  capsule 14 15 offset 17.6 0 0 fit 0.9 tags legs
collider calf_r   capsule 15 18 fit 0.9 tags legs
collider foot_r   capsule 18 length 11.7 radius 3.5 2.7 tags legs
collider heel_l   capsule 9 length 10.7 radius 4.6 4.6 offset 0 3.55 0 tags legs
collider heel_r   capsule 18 length 10.7 radius 4.6 4.6 offset 0 3.55 0 tags legs
collider arm_l    capsule 126 127 fit 0.85 side motion tags arms
collider forearm_l capsule 127 128 fit 0.85 side motion tags arms
collider hand_l   sphere 148 offset 8.2 0 0 radius 4.8 side rest tags hands left_hand
# the thumb, its root to past its tip
collider thumb_l  capsule 154 length 7 offset 1.4 0 0 fit 0.95 side rest tags hands left_hand
collider arm_r    capsule 264 265 fit 0.85 side motion tags arms
collider forearm_r capsule 265 266 fit 0.85 side motion tags arms
collider hand_r   sphere 286 offset 8.2 0 0 radius 4.8 side rest tags hands right_hand
collider thumb_r  capsule 292 length 7 offset 1.4 0 0 fit 0.95 side rest tags hands right_hand
# the head bone's own vertices (2156, his hair on bones of its own) stand -5.1 to 23.9 cm up it, 11.9 behind to 18.4
# in front, 14.6 from their middle (p90); the sphere on their middle
collider head     sphere 186 offset 9.4 3.2 0 radius 14 tags head
# the hair's alone: inside every root on the head (the nearest, 228, 11.7 cm from its centre), so none is pushed off its
# keyed place (Zexion's jitter came from roots inside it)
collider scalp    sphere 186 offset 8 1 0 radius 10.5 tags scalp
# the drawstrings' alone: over the chest's front, as demyx_h's (their roots 14.6 cm from 98-99's axis, his 13.8)
collider chest_strings capsule 98 99 radius 16.4 tags strings

# --- The lower coat: eight strips from the waist, one connected surface (axel_h's, the same bones) ---
# Each strip runs from just under the belt to the hem. 85 is shared: the front
# forks under it into two panels (88-90 and 92-94) with the slit between them.
group coat surface preset kh2_coat
  attach 2
  strip 26 29 30 31 32
  strip 34 37 38 39 40
  strip 42 45 46 47 48
  strip 50 53 54 55 56
  strip 58 61 62 63 64
  strip 66 69 70 71 72
  strip 74 77 78 79 80
  strip 82 85 88 89 90
  strip 85 92 93 94
  # the thighs carry the coat's shape: each control's target rides them by how near it
  # hangs to them at rest (up to 70%, none 34 cm away), so a wide stance spreads the coat
  # in its target and contact only catches what is left; a thigh coming up into it (a
  # crouch, a high step) carries the panels near it up to 90%. Not the shins: below the
  # knee the coat falls straight down off it rather than tracing the calf back
  follow_bone 5 radius 34 weight 0.9 always 0.7
  follow_bone 14 radius 34 weight 0.9 always 0.7
  topology mesh           # neighbours from the triangles skinned to them
  disconnect 88 92        # the front slit stays open
  # the back strips (26-32, 50-56, 74-80) ride the thighs less, the less the lower: the back hangs
  bone 29 follow_bone_scale 0.6
  bone 30 follow_bone_scale 0.4
  bone 31 follow_bone_scale 0.3
  bone 32 follow_bone_scale 0.2
  bone 53 follow_bone_scale 0.6
  bone 54 follow_bone_scale 0.4
  bone 55 follow_bone_scale 0.3
  bone 56 follow_bone_scale 0.2
  bone 77 follow_bone_scale 0.6
  bone 78 follow_bone_scale 0.4
  bone 79 follow_bone_scale 0.3
  bone 80 follow_bone_scale 0.2
  mode simulated
end

# --- Drawstrings: chains from the collar (99), hung at load: they stick straight forward at rest ---
group string_l chain preset kh2_string
  collide torso body strings
  attach 99
  chain 118 119 120 121
  hang 0 -1 0.35
  settle on steps 360
  mode simulated
end
group string_r chain preset kh2_string
  collide torso body strings
  attach 99
  chain 256 257 258 259
  hang 0 -1 0.35
  settle on steps 360
  mode simulated
end

# --- Hood: one bone (116) hanging behind the neck; it swings and lies on the back as one piece ---
group hood chain preset kh2_hood
  attach 99
  chain 116
  settle on steps 240
  mode simulated
end

# --- Hair on his head, from its end (187): single-bone spikes and the fringe's two-bone strands ---
group hair hair preset kh2_hair
  attach 187
  collide scalp
  tufts 189 191 193 195 197 199 201 212 214 216 218 220
  tuft 203 204
  tuft 206 207
  tuft 209 210
  tuft 222 223
  tuft 225 226
  tuft 228 229
end

# --- The locks by his face, from the neck's end (184): two strands, and two pieces whose forks (235-240, 246-251) ride
# them. Against the scalp alone: they rest on his chest and shoulder pieces as modelled (84 edges a frame through the
# body with no simulation, in the replay's idle); against the torso too they twitched 11.7 cm (0.04 without) ---
group side_hair hair preset kh2_hair
  attach 184
  collide scalp
  tufts 234 245
  tuft 231 232
  tuft 242 243
end

# --- The hair down his back (101 from the collar, not simulated): six short strands from its end (102) ---
group back_hair hair preset kh2_hair
  attach 102
  collide torso
  tufts 104 106 108 110 112 114
end

# --- Animation controls ---
# B_EX170's own motions carry its coat and drawstrings as KH2 animated them: there they guide the cloth, and the
# thighs carry none of it. The game's own animations keep the cloth on the hips and thighs.
control own_motions
  animation *
  slot 0-255
  groups coat string_l string_r
  mode guided
  follow_frequency 6 hem 3
  follow_damping 0.8
  follow_bone_scale 0
  blend 0.15 0.15
end
