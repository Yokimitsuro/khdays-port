# Secondary motion of KH2's Axel (B_EX110) drawn as Days' axel_h and axel: roxas_bf_h's coat and drawstrings on his
# bones, with his hood and hair. Bones are KH2 bone numbers (kh2_<n>). docs/SECONDARY_MOTION.md explains every line.
secondary_motion 1
units cm
include kh2_secondary_presets.txt

# 60 Hz; 3 substeps of 2 passes, as roxas_bf_h
simulation rate 60 substeps 3 iterations 2
gravity 0 -9.81 0           # m/s^2, the world's y up
environment on thickness 2 ground  # the room's floors alone
teleport 150 100            # a jump of 1.5 m or 100 degrees in one tick rebases
first_pose_iterations 30

# --- Body colliders (KH2 bones): `fit` sizes itself to the vertices skinned to them. The fixed sizes are roxas_bf_h's,
# scaled by his parts against Roxas's: thighs 1.3 as long, pelvis 1.5, feet 0.77 as wide and the heel 6.4 cm behind
# the ankle against 8.6, hands 1.1 as long, head 1.3 as tall. ---
collider pelvis   capsule 2 length 26.8 fit 0.85 tags body
collider spine    capsule 96 99 fit 0.85 tags torso
collider chest    capsule 99 length 22.3 fit 0.85 tags torso
# the thighs sized on their thinned meshes (front/back 10-11 cm, 0.9 of it; fitted they stood 11.5)
collider thigh_l  capsule 5 6 offset 15.7 0 0 radius 10.0 8.5 tags legs
collider calf_l   capsule 6 9 fit 0.9 tags legs
collider foot_l   capsule 9 length 11.7 radius 3.5 2.7 tags legs
collider thigh_r  capsule 14 15 offset 15.7 0 0 radius 10.0 8.5 tags legs
collider calf_r   capsule 15 18 fit 0.9 tags legs
collider foot_r   capsule 18 length 11.7 radius 3.5 2.7 tags legs
collider heel_l   capsule 9 length 10.7 radius 4.6 4.6 offset 0 3.55 0 tags legs
collider heel_r   capsule 18 length 10.7 radius 4.6 4.6 offset 0 3.55 0 tags legs
collider arm_l    capsule 112 113 fit 0.85 side motion tags arms
collider forearm_l capsule 113 114 fit 0.85 side motion tags arms
collider hand_l   sphere 119 offset 8.2 0 0 radius 4.8 side rest tags hands left_hand
# the thumb, its root to past its tip
collider thumb_l  capsule 125 length 7 offset 1.4 0 0 fit 0.95 side rest tags hands left_hand
collider arm_r    capsule 199 200 fit 0.85 side motion tags arms
collider forearm_r capsule 200 201 fit 0.85 side motion tags arms
collider hand_r   sphere 206 offset 8.2 0 0 radius 4.8 side rest tags hands right_hand
collider thumb_r  capsule 212 length 7 offset 1.4 0 0 fit 0.95 side rest tags hands right_hand
collider head     sphere 161 offset 8 1 0 radius 12 tags head
# the hair's alone: 184, the crown, turns about the forehead, in the head's sphere; from radius 10.5 it was pushed
# up and forward off its keyed place (5.2 cm at 12) in every pose (replay, idle)
collider scalp    sphere 161 offset 8 1 0 radius 10 tags scalp
# the drawstrings' alone: over the chest's front (z 10.6, the axis about -3.5), 1.4 clear of it as on roxas_bf_h
collider chest_strings capsule 99 100 radius 15.5 tags strings

# --- The lower coat: eight strips from the waist, one connected surface ---
# Each strip runs from just under the belt to the hem. 85 is shared: the front
# forks under it into two panels (88-90 and 92-94) with the slit between them.
group coat surface preset kh2_coat
  # replay on axel_h, the game's own animations: swings held closer (the 2-hit attack stretched the strips 411%, 66% held)
  # and the drawn shape eased less, so the guard pulling lagging faces off the legs jumps less (B002 82->27 cm,
  # guard 51->37, run 27->19 cm p99)
  swing_limit 40 hem 55
  smoothing 0.05
  attach 2
  strip 26 29 30 31 32
  strip 34 37 38 39 40
  strip 42 45 46 47 48
  strip 50 53 54 55 56
  strip 58 61 62 63 64
  strip 66 69 70 71 72
  strip 74 77 78 79 80
  strip 82 85 88 89 90
  strip 85 92 93 94
  # the thighs carry the coat's shape: each control's target rides them by how near it
  # hangs to them at rest (up to 70%, none 30 cm away), so a wide stance spreads the coat
  # in its target and contact only catches what is left; a thigh coming up into it (a
  # crouch, a high step) carries the panels near it up to 90%. Not the shins: below the
  # knee the coat falls straight down off it rather than tracing the calf back
  # (his coat stands 1.14 as far from the legs as Roxas's: radius 34 for roxas_bf_h's 30)
  follow_bone 5 radius 34 weight 0.9 always 0.7
  follow_bone 14 radius 34 weight 0.9 always 0.7
  topology mesh           # neighbours from the triangles skinned to them
  disconnect 88 92        # the front slit stays open
  # the back strips (26-32, 50-56, 74-80) ride the thighs less, the less the lower: the back
  # hangs. `follow_bone_scale` can also be set by group (with `hem`) or per animation (control)
  bone 29 follow_bone_scale 0.6
  bone 30 follow_bone_scale 0.4
  bone 31 follow_bone_scale 0.3
  bone 32 follow_bone_scale 0.2
  bone 53 follow_bone_scale 0.6
  bone 54 follow_bone_scale 0.4
  bone 55 follow_bone_scale 0.3
  bone 56 follow_bone_scale 0.2
  bone 77 follow_bone_scale 0.6
  bone 78 follow_bone_scale 0.4
  bone 79 follow_bone_scale 0.3
  bone 80 follow_bone_scale 0.2
  mode simulated          # P_EX110's motions carry no coat tracks
  # `no_sim`: the coat not simulated, each control on its target (the hips, the thighs' carry, the drape
  # round the legs) every tick; here for the whole coat, or in a `control` for some animations
  # no_sim
  # `clip_world`: the room inside the coat's closed volume not drawn, so a wall it goes through shows the
  # coat's inside, not the wall; 2 mm outside its drawn controls (on them, the skin's blending left a line
  # of floor showing where it met the coat's inside)
  # off in gameplay (the room drawn over the coat as it is); the cutscene coat (e_rox_hb) keeps it
  # clip_world inset -0.2
end

# --- Drawstrings: chains from the collar (100), hung at load: B_EX110's stick straight forward at rest ---
group string_l chain preset kh2_string
  collide torso body strings
  attach 100
  chain 104 105 106 107
  hang 0 -1 0.35
  settle on steps 360
  mode simulated
end
group string_r chain preset kh2_string
  collide torso body strings
  attach 100
  chain 191 192 193 194
  hang 0 -1 0.35
  settle on steps 360
  mode simulated
end

# --- Hood: one bone (102) hanging behind the neck; it swings and lies on the back as one piece ---
group hood chain preset kh2_hood
  attach 100
  chain 102
  settle on steps 240
  mode simulated
end

# --- Hair: single-bone spikes on the head's end (162) ---
group hair hair preset kh2_hair
  attach 162
  collide scalp
  tufts 164 166 168 170 172 174 176 178 180 182 184 186
end

# --- Animation controls ---
# B_EX110's own motions carry its coat and drawstrings as KH2 animated them: there they guide the cloth, and the thighs
# carry none of it. The game's own animations keep the cloth on the hips and thighs, as roxas_bf_h's do.
control own_motions
  animation *
  slot 0-255
  groups coat string_l string_r
  mode guided
  follow_frequency 6 hem 3
  follow_damping 0.8
  follow_bone_scale 0
  blend 0.15 0.15
end
