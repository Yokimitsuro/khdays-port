# Secondary motion of KH2's Saix (B_EX160) drawn as Days' saix_h and saix: axel_h's coat on the same bones, his
# drawstrings, hood and hair. Bones are KH2 bone numbers (kh2_<n>). docs/SECONDARY_MOTION.md explains every line. His
# cuffs are fitted in bones.txt (`cuff`), not simulated.
secondary_motion 1
units cm
include kh2_secondary_presets.txt

# 60 Hz; 3 substeps of 2 passes, as roxas_bf_h
simulation rate 60 substeps 3 iterations 2
gravity 0 -9.81 0           # m/s^2, the world's y up
environment on thickness 2 ground  # the room's floors alone
teleport 150 100            # a jump of 1.5 m or 100 degrees in one tick rebases
first_pose_iterations 30

# --- Body colliders (KH2 bones): `fit` sizes itself to the vertices skinned to them. His legs, hands and pelvis are
# luxord_h's vertex for vertex (demyx_h's), so the fixed sizes are theirs (axel_h's); the pelvis, chest and thighs by
# his own bones' lengths. ---
collider pelvis   capsule 2 length 22.1 fit 0.85 tags body
collider spine    capsule 96 98 fit 0.85 tags torso
# 21.5 of the 27.6 from 98 to the collar (99), as demyx_h's 18 of 23.2: at its full length its end reaches the hood
collider chest    capsule 98 length 21.5 fit 0.85 tags torso
collider thigh_l  capsule 5 6 offset 17.8 0 0 fit 0.8 tags legs
collider calf_l   capsule 6 9 fit 0.9 tags legs
collider foot_l   capsule 9 length 11.7 radius 3.5 2.7 tags feet
collider thigh_r  capsule 14 15 offset 17.8 0 0 fit 0.8 tags legs
collider calf_r   capsule 15 18 fit 0.9 tags legs

collider heel_l   capsule 9 length 10.7 radius 4.6 4.6 offset 0 3.55 0 tags feet
collider heel_r   capsule 18 length 10.7 radius 4.6 4.6 offset 0 3.55 0 tags feet

collider forearm_l capsule 112 113 fit 0.85 side motion tags arms

collider thumb_l  capsule 139 length 7 offset 1.4 0 0 fit 0.95 side rest tags hands left_hand
collider arm_r    capsule 324 325 fit 0.85 side motion tags arms

# the head bone's own vertices (2934, his hair on bones of its own) stand -4.1 to 25.5 cm up it, 15.8 behind to 13.2
# in front, 15 from their middle (p90); the sphere on their middle
collider head     sphere 169 offset 10.7 -1.3 0 radius 14 tags head
# the hair's alone: inside every simulated root (the nearest, the sideburns 210 and 269, 10.2 cm from its centre), so
# none is pushed off its keyed place (Zexion's jitter came from roots inside it)
collider scalp    sphere 169 offset 8 1.5 0 radius 9 tags scalp
# the drawstrings' alone: over the chest's front, as demyx_h's (their roots 13.8 cm from 98-99's axis, his too)
collider chest_strings capsule 98 99 radius 15.5 tags strings

# --- The lower coat: eight strips from the waist, one connected surface (axel_h's, the same bones) ---
# Each strip runs from just under the belt to the hem. 85 is shared: the front
# forks under it into two panels (88-90 and 92-94) with the slit between them.
group coat surface preset kh2_coat
  attach 2
  strip 26 29 30 31 32
  strip 34 37 38 39 40
  strip 42 45 46 47 48
  strip 50 53 54 55 56
  strip 58 61 62 63 64
  strip 66 69 70 71 72
  strip 74 77 78 79 80
  strip 82 85 88 89 90
  strip 85 92 93 94
  # the thighs carry the coat's shape: each control's target rides them by how near it
  # hangs to them at rest (up to 70%, none 34 cm away), so a wide stance spreads the coat
  # in its target and contact only catches what is left; a thigh coming up into it (a
  # crouch, a high step) carries the panels near it up to 90%. Not the shins: below the
  # knee the coat falls straight down off it rather than tracing the calf back
  follow_bone 5 radius 34 weight 0.9 always 0.7
  follow_bone 14 radius 34 weight 0.9 always 0.7
  topology mesh           # neighbours from the triangles skinned to them
  disconnect 88 92        # the front slit stays open
  # the back strips (26-32, 50-56, 74-80) ride the thighs less, the less the lower: the back hangs
  bone 29 follow_bone_scale 0.6
  bone 30 follow_bone_scale 0.4
  bone 31 follow_bone_scale 0.3
  bone 32 follow_bone_scale 0.2
  bone 53 follow_bone_scale 0.6
  bone 54 follow_bone_scale 0.4
  bone 55 follow_bone_scale 0.3
  bone 56 follow_bone_scale 0.2
  bone 77 follow_bone_scale 0.6
  bone 78 follow_bone_scale 0.4
  bone 79 follow_bone_scale 0.3
  bone 80 follow_bone_scale 0.2
  mode simulated
end

# --- The two long locks by his face, drawstrings as xaldin_h's braids (user): each mane's long run from the head
# (223, 225, 235, 238: 10.6, 14.7, 12 and 5 cm, down to his chest; 282, 284, 294, 297 the other), its zero-length
# helpers between carried, its spikes (226-232, 241-252; 285-291, 300-311) carried by the bones they branch from.
# Against his scalp and chest, with xaldin_h's braids' inertia, damping and swing ---
group locks chain preset kh2_string
  collide scalp torso
  thickness 1.5
  inertia 0.3
  damping 4
  swing_limit 180
  attach 170
  chain 223 225 235 238
  chain 282 284 294 297
  settle on steps 360
  mode simulated
end

# --- Animation controls ---
# B_EX160's own motions carry its coat and drawstrings as KH2 animated them: there they guide the cloth, and the
# thighs carry none of it. The game's own animations keep the cloth on the hips and thighs.
control own_motions
  animation *
  slot 0-255
  groups coat string_l string_r
  mode guided
  follow_frequency 6 hem 3
  follow_damping 0.8
  follow_bone_scale 0
  blend 0.15 0.15
end

# His KH2 motions key his locks: there they follow them, as xaldin_h's braids follow his hover. The game's own
# animations leave them simulated
control own_locks
  animation *
  slot 0-511
  groups locks
  mode guided
  follow_frequency 6
  follow_damping 0.7
  blend 0.15 0.15
end
